    3.2- Script completo

    #!/bin/bash

echo "🔒 Aplicando política de contraseñas: caducidad cada 30 días, longitud mínima de 8 caracteres y verificación de complejidad..."

# Expiración de la contraseña cada 30 días
samba-tool domain passwordsettings set --max-pwd-age=30

# Activar la complejidad de contraseña (requiere mayúsculas, minúsculas y números)
samba-tool domain passwordsettings set --complexity=on

# Establecer una longitud mínima de 8 caracteres
samba-tool domain passwordsettings set --min-pwd-length=8

# Evitar que los usuarios reutilicen sus últimas 5 contraseñas
samba-tool domain passwordsettings set --history-length=5

echo "✅ Políticas de contraseña aplicadas correctamente."



# Listas de usuarios y grupos que deseas crear
read -p "Introduce los nombres de los grupos, separados por espacios: " -a grupos

# Crear los grupos en el dominio si no existen
for grupo in "${grupos[@]}"; do
    if samba-tool group list | grep -q "^$grupo$"; then
        echo "El grupo $grupo ya existe en el dominio."
    else
        echo "Creando grupo $grupo en el dominio..."
        samba-tool group add "$grupo"
    fi
done

# Preguntar el número de usuarios que queremos crear
read -p "¿Cuántos usuarios deseas crear? " num_usuarios

# Crear usuarios y asignarlos a grupos
for ((i = 1; i <= num_usuarios; i++)); do
    read -p "Introduce el nombre del usuario $i: " usuario

    # Si el usuario ya existe, lo informamos y saltamos la creación
    if samba-tool user show "$usuario" &>/dev/null; then
        echo "El usuario $usuario ya existe en el dominio."
    else
        # Bucle para solicitar la contraseña hasta que cumpla la política
        creado=0
        while [ $creado -eq 0 ]; do
            read -p "Introduce la contraseña para el usuario $usuario: " samba_pass
            # Intentamos crear el usuario; si falla (por ejemplo, por política de contraseñas) se repite el bucle.
            if samba-tool user add "$usuario" "$samba_pass"; then
                echo ✅ "Usuario $usuario creado correctamente."
                creado=1
            else
                echo "La contraseña no cumple con la política de contraseñas. Por favor, inténtalo de nuevo."
            fi
        done
    fi

    # Asignar el usuario a un grupo específico
    echo "Selecciona el grupo del dominio para $usuario:"
    select grupo in "${grupos[@]}"; do
        if [[ -n "$grupo" ]]; then
            echo "Asignando $usuario al grupo $grupo..."
            samba-tool group addmembers "$grupo" "$usuario"
            break
        else
            echo "Selección no válida, intenta de nuevo."
        fi
    done

    # Asignar privilegios de administrador si se requiere
    read -p "¿Deseas asignar privilegios de administrador a $usuario? (s/n): " admin
    if [[ "$admin" =~ ^[sS]$ ]]; then
        echo "Asignando privilegios de administrador a $usuario..."
        samba-tool group addmembers "Domain Admins" "$usuario"
    else
        echo ✅ "Asignando permisos de usuario a $usuario..."
    fi
done

# Reiniciar el servicio de Samba para aplicar cambios
systemctl restart samba-ad-dc
echo ✅ "Usuarios y grupos creados exitosamente en Samba."
